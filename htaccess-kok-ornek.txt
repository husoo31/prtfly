# İSTEĞE BAĞLI: Alan adınızı public/ klasörüne yönlendiremiyorsanız bu dosyayı
# kök dizinde ".htaccess" olarak yeniden adlandırın. Belge kökü public/ ise KULLANMAYIN;
# bazı sunucular (ör. Hestia) bu dosyadaki yönergelere izin vermez ve 500 hatası verir.
<IfModule mod_rewrite.c>
    RewriteEngine On

    RewriteRule ^(\.env.*|\.git.*|composer\.(json|lock)|artisan|phpunit\.xml)$ - [F,L,NC]
    RewriteRule ^(app|bootstrap|config|database|lang|resources|routes|storage|tests|vendor)(/.*)?$ - [F,L,NC]

    RewriteCond %{REQUEST_URI} !(^|/)public/
    RewriteRule ^(.*)$ public/$1 [L]
</IfModule>

<FilesMatch "^\.">
    Require all denied
</FilesMatch>
